### Cell 2 — synthetic Tycho DEM + Lommel-Seeliger renderer (self-contained, no upload needed)
```python
import numpy as np
import torch, torch.nn as nn, torch.nn.functional as F
import os

def make_crater_dem(size=512, rim_radius=0.42, depth=1900.0, rim_height=350.0,
                     central_peak_h=2000.0, central_peak_r=0.09, seed=7):
    rng = np.random.default_rng(seed)
    y, x = np.mgrid[0:size, 0:size].astype(np.float32)
    cx, cy = size/2, size/2
    r = np.sqrt((x-cx)**2 + (y-cy)**2) / (size/2)
    bowl = -depth * np.clip(1 - (r/rim_radius)**2, 0, 1)
    rim = rim_height * np.exp(-((r-rim_radius)**2) / (2*0.03**2))
    peak = central_peak_h * np.exp(-(r**2) / (2*central_peak_r**2))
    dem = bowl + rim + peak
    from PIL import Image
    rough = np.zeros((size, size), dtype=np.float32)
    for octave, amp in [(4,40),(8,18),(16,8),(32,3)]:
        noise = rng.standard_normal((octave, octave)).astype(np.float32)
        noise_img = np.array(Image.fromarray(noise).resize((size,size), Image.BICUBIC))
        rough += amp * noise_img
    return (dem + rough).astype(np.float32)

DEM = make_crater_dem()
np.save("/kaggle/working/tycho_dem.npy", DEM)   # keep a copy of what you trained on
print("DEM shape:", DEM.shape, "min/max:", DEM.min(), DEM.max())
```

### Cell 3 — triplet generator + Dataset
```python
def lommel_seeliger(albedo, cos_i, cos_e):
    return albedo * cos_i / (cos_i + cos_e + 1e-8)

def render_shaded(patch, az_deg, el_deg, albedo=0.12):
    gy, gx = np.gradient(patch)
    normal = np.dstack([-gx, -gy, np.ones_like(patch)]).astype(np.float32)
    normal /= (np.linalg.norm(normal, axis=2, keepdims=True) + 1e-8)
    az, el = np.radians(az_deg), np.radians(el_deg)
    sun = np.array([np.cos(el)*np.cos(az), np.cos(el)*np.sin(az), np.sin(el)], dtype=np.float32)
    cos_i = np.clip(normal @ sun, 0, 1)
    b = lommel_seeliger(albedo, cos_i, np.ones_like(cos_i))
    return b / (b.max() + 1e-8)

def random_crop(dem, size, rng):
    h, w = dem.shape
    y, x = rng.integers(0, h-size), rng.integers(0, w-size)
    return dem[y:y+size, x:x+size]

def make_triplet(dem, size, rng):
    anchor_patch = random_crop(dem, size, rng)
    anchor = render_shaded(anchor_patch, rng.uniform(0,360), rng.uniform(15,40))
    positive = render_shaded(anchor_patch, rng.uniform(0,360), rng.uniform(40,70))
    neg_patch = random_crop(dem, size, rng)
    negative = render_shaded(neg_patch, rng.uniform(0,360), rng.uniform(15,70))
    return anchor, positive, negative

class TripletDataset(torch.utils.data.Dataset):
    def __init__(self, dem, n_pairs=4000, patch_size=128, seed=0):
        self.dem, self.n, self.patch_size = dem, n_pairs, patch_size
        self.rng = np.random.default_rng(seed)
    def __len__(self): return self.n
    def __getitem__(self, idx):
        a, p, n = make_triplet(self.dem, self.patch_size, self.rng)
        t = lambda z: torch.from_numpy(z[None,:,:].astype(np.float32))
        return t(a), t(p), t(n)

ds = TripletDataset(DEM, n_pairs=4000, patch_size=128)
dl = torch.utils.data.DataLoader(ds, batch_size=32, shuffle=True, num_workers=2)
```

### Cell 4 — model
```python
class PatchEncoder(nn.Module):
    def __init__(self, embed_dim=128):
        super().__init__()
        self.net = nn.Sequential(
            nn.Conv2d(1,16,3,padding=1), nn.ReLU(), nn.MaxPool2d(2),
            nn.Conv2d(16,32,3,padding=1), nn.ReLU(), nn.MaxPool2d(2),
            nn.Conv2d(32,64,3,padding=1), nn.ReLU(), nn.MaxPool2d(2),
            nn.Conv2d(64,128,3,padding=1), nn.ReLU(), nn.AdaptiveAvgPool2d(1),
        )
        self.fc = nn.Linear(128, embed_dim)
    def forward(self, x):
        return F.normalize(self.fc(self.net(x).flatten(1)), dim=1)

def triplet_loss(a, p, n, margin=0.2):
    d_pos = (a-p).pow(2).sum(1).sqrt()
    d_neg = (a-n).pow(2).sum(1).sqrt()
    return torch.clamp(d_pos - d_neg + margin, min=0).mean()

DEVICE = "cuda" if torch.cuda.is_available() else "cpu"
model = PatchEncoder().to(DEVICE)
opt = torch.optim.Adam(model.parameters(), lr=1e-3)
print("device:", DEVICE)
```

### Cell 5 — training loop
```python
EPOCHS = 30
for epoch in range(EPOCHS):
    model.train()
    total = 0.0
    for a, p, n in dl:
        a, p, n = a.to(DEVICE), p.to(DEVICE), n.to(DEVICE)
        opt.zero_grad()
        ea, ep, en = model(a), model(p), model(n)
        loss = triplet_loss(ea, ep, en)
        loss.backward(); opt.step()
        total += loss.item() * a.size(0)
    if epoch % 5 == 0 or epoch == EPOCHS-1:
        print(f"epoch {epoch:3d}  loss {total/len(ds):.4f}")
```

### Cell 6 — validation (ranking accuracy, not a fixed-vector assertion)
```python
def validate(model, dem, n_trials=200, patch_size=128):
    model.eval()
    correct = 0
    rng = np.random.default_rng(999)
    with torch.no_grad():
        for _ in range(n_trials):
            a, p, n = make_triplet(dem, patch_size, rng)
            t = lambda z: torch.from_numpy(z[None,None,:,:].astype(np.float32)).to(DEVICE)
            ea, ep, en = model(t(a)), model(t(p)), model(t(n))
            d_pos = (ea-ep).pow(2).sum().sqrt()
            d_neg = (ea-en).pow(2).sum().sqrt()
            correct += (d_pos < d_neg).item()
    return correct / n_trials

acc = validate(model, DEM)
print(f"triplet ranking accuracy: {acc:.3f} (target: comfortably > 0.5, ideally > 0.9)")
```
